TESTABLE WORKFLOW CODE

Readable on day one. Testable on every change.

Define reusable components as task classes, then run the same named workflow from tests, local development, package scripts, and CI.

↳
Designed for tests

Use bz.create() for isolation, inject configuration, and assert workflow behavior through the same public API used in production.

beelzebub.ts 
import bz from 'beelzebub';

class BuildTasks extends bz.Tasks {
  clean() {
    this.logger.log('clean');
  }

  async compile() {
    await compileProject();
  }
}

bz.add(BuildTasks);
await bz.run(
  'BuildTasks.clean',
  'BuildTasks.compile'
);
class ReleaseTasks extends bz.Tasks {
  verify() {
    return this.$parallel(
      '.lint',
      '.test'
    );
  }

  release() {
    return this.$sequence(
      '.verify',
      '.build',
      '.publish'
    );
  }
}
import {
  BzTasks, defaultTask, help, vars
} from 'beelzebub';

class DeployTasks extends BzTasks {
  @defaultTask
  @help('Deploy the application')
  @vars({
    env: { type: 'string', required: true }
  })
  deploy(options: { env: string }) {
    this.logger.info(options.env);
  }
}
